cmake_minimum_required(VERSION 3.16)

set(ASSETS_DIR "${CMAKE_SOURCE_DIR}/assets" CACHE PATH "Directory containing assets")
set(target_name underice)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BIN_DIR "${CMAKE_SOURCE_DIR}/bin" CACHE PATH "Where the .bin goes")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${BIN_DIR})

set(ZOS_TOOLCHAIN sdcc)
include($ENV{ZOS_PATH}/cmake/zos_init.cmake)

zos_use(zvb)
include($ENV{ZGDK_PATH}/cmake/zgdk_init.cmake)

project(underice C ASM)

add_executable(${target_name}
    src/controls.c
    src/main.c
    src/stream.c
    src/sea.c
    src/seal.c
    src/actors.c
    src/hud.c
    src/sfx.c
    src/music.c
    src/assets.c
)
target_compile_options(${target_name} PRIVATE --opt-code-size)
# the sea is 94 rows deep (SEA_ROWS): the core's tile flag row table has to cover it


# For recording videos: a bot plays (src/autopilot.c), and the game can start on a later dive
option(AUTOPILOT "Bot plays the game (video recording)" OFF)
if(AUTOPILOT)
    target_sources(${target_name} PRIVATE src/autopilot.c)
    target_compile_definitions(${target_name} PRIVATE AUTOPILOT=1)
endif()
set(START_DIVE 0 CACHE STRING "First dive, 0-based (video recording)")
target_compile_definitions(${target_name} PRIVATE START_DIVE=${START_DIVE})

find_package(ZVB_SDK REQUIRED CONFIG PATHS "$ENV{ZVB_SDK_PATH}/cmake" NO_DEFAULT_PATH)
find_package(ZGDK REQUIRED CONFIG PATHS "$ENV{ZGDK_PATH}/cmake" NO_DEFAULT_PATH)
target_link_libraries(${target_name} PUBLIC zgdk zvb_gfx zvb_sound)

# tileset.gif is generated by tools/make_assets.py (committed, so the build doesn't need Pillow).
# LZ instead of RLE: ZVB's gif2zeal() COMPRESSED flag only passes "-z" (RLE), so this calls
# gif2zeal.py the same way with "-z lz" (from Bubble Keep).
file(GLOB GIF_SRCS CONFIGURE_DEPENDS "${ASSETS_DIR}/*.gif")
foreach(gif ${GIF_SRCS})
    get_filename_component(gif_we ${gif} NAME_WE)
    set(gif_out ${CMAKE_BINARY_DIR}/assets)
    add_custom_command(
        OUTPUT ${gif_out}/${gif_we}.zts ${gif_out}/${gif_we}.ztp
        COMMAND ${CMAKE_COMMAND} -E make_directory ${gif_out}
        COMMAND ${Python3_EXECUTABLE} $ENV{ZVB_SDK_PATH}/tools/zeal2gif/gif2zeal.py
                -i ${gif} -t ${gif_out}/${gif_we}.zts -p ${gif_out}/${gif_we}.ztp -o ${gif_out} -z lz
        DEPENDS ${gif} $ENV{ZVB_SDK_PATH}/tools/zeal2gif/gif2zeal.py
        COMMENT "Processing asset ${gif_we}.gif (LZ)"
        VERBATIM
    )
    add_custom_target(${target_name}_${gif_we}_asset ALL DEPENDS ${gif_out}/${gif_we}.zts ${gif_out}/${gif_we}.ztp)
    add_dependencies(${target_name} ${target_name}_${gif_we}_asset)
endforeach()
file(RELATIVE_PATH BUILD_REL ${CMAKE_SOURCE_DIR} ${CMAKE_BINARY_DIR})
# tileset + music (the .zmt tracks are written by tools/make_music.py, committed).
# Bot builds only embed the dive track (see music.c).
set(MUSIC_SRCS assets/dive.zmt)
if(NOT AUTOPILOT)
    list(APPEND MUSIC_SRCS assets/title.zmt assets/clear.zmt assets/over.zmt)
endif()
embed_assets(${target_name} BUILD_DIR ${BUILD_REL}/assets FILES ${GIF_SRCS} ${MUSIC_SRCS})

zos_add_outputs(${target_name})

set_property(TARGET ${target_name} APPEND PROPERTY LINK_DEPENDS "$ENV{ZGDK_PATH}/lib/zgdk.lib")
